```python
from pathlib import Path
import hashlib
root = Path(".")
paths = [root / "SKILL.md", *sorted((root / "references").glob("*.md")),
         *sorted((root / "assets/templates").glob("*.md")),
         *sorted((root / "assets/examples").glob("*.md"))]
digest = hashlib.sha256()
for path in paths:
    digest.update((path.relative_to(root).as_posix() + chr(0) +
                   path.read_text(encoding="utf-8")).encode())
print(digest.hexdigest())
```
